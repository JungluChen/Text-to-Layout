# Anchored lexer must consume every character, including malformed quote prefixes.
TOKEN=re.compile(r'\s+|\(|\)|"(?:\\["\\]|[^"\\])*"|[^\s()"\\]+')
def parse(t):
 stack=[];roots=[];pos=0
 while pos<len(t):
  m=TOKEN.match(t,pos)
  if m is None:raise ValueError('invalid token at '+str(pos))
  tok=m.group();pos=m.end()
  if tok.isspace():continue
  if tok=='(':
   if len(stack)>=32:raise ValueError('depth')
   node=[]
   if stack:stack[-1].append(node)
   else:roots.append(node)
   stack.append(node)
  elif tok==')':
   if not stack:raise ValueError('close')
   stack.pop()
  else:
   if not stack:raise ValueError('atom outside root')
   stack[-1].append(tok)
 if stack or len(roots)!=1:raise ValueError('root')
 return roots[0]
