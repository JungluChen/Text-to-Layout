from pathlib import Path
import json,re,hashlib
p=Path('/private/tmp/qeda-geometry-probe');ref=json.loads((p/'reference.json').read_text());results=[]
for i in [1,2,3,4]:
 d=p/f'run-{i}';f=next((d/'kicad/probe.pretty').glob('*.kicad_mod'));t=f.read_text();depth=0;minimum=0;roots=0
 for token in re.findall(r'"(?:\\.|[^"\\])*"|[()]',t):
  if token=='(':
   if depth==0:roots+=1
   depth+=1
  elif token==')':depth-=1;minimum=min(minimum,depth)
 pads={}
 for m in re.finditer(r'\(pad (\d+) smd rect \(at ([\d.-]+) ([\d.-]+)\) \(size ([\d.-]+) ([\d.-]+)\)',t):
  pads[m[1]]=dict(zip(['x','y','width','height'],map(float,m.groups()[1:])))
 err=max(abs(pads[n][k]-v) for n,pad in ref['pads'].items() for k,v in pad.items())
 symbol=(d/'kicad/probe.lib').read_text();pins=[x for x in symbol.splitlines() if x.startswith('X ')]
 results.append({'run':i,'pad_count':len(pads),'pad_max_error_mm':err,'pad_reference_pass':pads.keys()==ref['pads'].keys() and err<=ref['absolute_tolerance_mm'],'sexpr_final_depth':depth,'sexpr_min_depth':minimum,'sexpr_root_count':roots,'sexpr_balance_pass':depth==0 and minimum==0 and roots==1,'symbol_pin_records':pins,'acceptance':'REJECTED','hashes':{str(f.relative_to(d)):hashlib.sha256(f.read_bytes()).hexdigest() for f in sorted((d/'kicad').rglob('*')) if f.is_file()}})
(p/'assessment.json').write_text(json.dumps(results,indent=2)+'\n');print(json.dumps([{k:v for k,v in x.items() if k!='hashes'} for x in results],indent=2))
