from pathlib import Path
import json,subprocess,hashlib
p=Path('/private/tmp/qeda-geometry-probe')
reference={'units':'mm','pads':{'1':{'x':-1.5,'y':0.,'width':1.,'height':1.5},'2':{'x':1.5,'y':0.,'width':1.,'height':1.5}},'absolute_tolerance_mm':1e-6,'rationale':'Explicit custom geometry uses exactly representable values at default 0.001 mm output resolution; 1e-6 mm permits only floating parsing roundoff, not a quantization shift.','scope':'Pad geometry and numbering only; no IPC, fabrication or electrical claim.'}

yaml='''name: ProbeResistor
keywords: resistor

  L: 1
  R: 2

  symbol: resistor

 print([str(f.relative_to(d)) for f in d.rglob('*') if f.is_file()])
