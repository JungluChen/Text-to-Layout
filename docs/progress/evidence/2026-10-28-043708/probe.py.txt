from pathlib import Path
from dataclasses import asdict
import hashlib,json,os
from textlayout.external.qeda_inventory import compare_inventories
expected=Path('docs/progress/evidence/2026-10-26-102228/runtime-before.json').read_bytes()
results={}
for name,root in [('prior',Path('/private/tmp/qeda-runtime-audit/runtime')),('reconstructed',Path('/private/tmp/qeda-closure-audit/reconstructed'))]:
 rows=[]
 for p in sorted((root/'node_modules').rglob('*')):
  if p.is_symlink():rows.append({'path':p.relative_to(root).as_posix(),'symlink':os.readlink(p)})
  elif p.is_file():rows.append({'path':p.relative_to(root).as_posix(),'sha256':hashlib.sha256(p.read_bytes()).hexdigest()})
 observed=json.dumps(rows,indent=2).encode()
 results[name]=asdict(compare_inventories(expected,observed))
print(json.dumps(results,indent=2))
