import runpy,tempfile
from pathlib import Path
from unittest.mock import patch
ns=runpy.run_path('tests/textlayout_suite/test_palace_ci_observer.py')
original=Path.write_text
def windows_write(self,data,*args,**kwargs):
    kwargs['newline']='\r\n'
    return original(self,data,*args,**kwargs)
for i in range(2):
    with tempfile.TemporaryDirectory() as tmp,patch.object(Path,'write_text',windows_write):
        try:
            ns['test_cgroup_membership_selects_own_group'](Path(tmp))
        except AssertionError:
            b=(Path(tmp)/'groups/job/memory.current').read_bytes()
            print(f'Run {i+1}: FAIL; fixture bytes {b!r}, assertion expects LF')
        else:
            print(f'Run {i+1}: PASS')
